cmake_minimum_required(VERSION 3.20)
project(LumenSSPT LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows x64 with MSVC is required")
endif()
add_executable(LumenSSPT native/main.cpp)
target_compile_features(LumenSSPT PRIVATE cxx_std_17)
target_compile_options(LumenSSPT PRIVATE /O2 /W4 /EHsc)
target_link_libraries(LumenSSPT PRIVATE d3d12 dxgi dxguid)
set_target_properties(LumenSSPT PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}")
